/** NIMEX Check-in Daily — Apps Script (วางใน Extensions > Apps Script ของ Google Sheet ใหม่ บันทึกเป็น Code.gs)
 *  Script Properties: ADMIN_KEY = รหัสผู้ดูแล (ห้ามเขียนรหัสลงไฟล์นี้)
 *  Deploy > New deployment > Web app > Execute as: Me · Who has access: Anyone → คัดลอก URL /exec ใส่ใน config.js
 *  แท็บ Rows / Settings / Log สร้างให้อัตโนมัติ
 *  - Rows     : ผลการ Check-in รายคันรายวัน (คำนวณจากหน้าเว็บ สูตรเดิม) ทุกช่องเป็นข้อความ
 *  - Settings : ค่าตั้งที่ใช้ร่วมกัน (Config ค่าปรับราย DC, ประเภทรถ/งาน, Waive, เปิด/ปิดค่าปรับ, เกรด/อินโฟกราฟิก)
 *               ค่าที่ยาวจะแบ่งเป็นหลายแถว (part) เพราะเซลล์เก็บได้ไม่เกิน 50,000 ตัวอักษร
 *  - บันทึกข้อมูล: begin → append (หลายครั้ง) → finish  เขียนลงแท็บชั่วคราวก่อน แล้วสลับเป็น Rows ตอน finish
 *    ถ้าส่งไม่ครบ ข้อมูลเดิมใน Rows ยังอยู่ครบ */
var COLS = ['dc','code','plate','vehicleType','vehicleKind','dateISO','configStart','checkinTime','diffMin','distanceKm','status','fine','waived','waivedFine','billDelivered','dropTotal','dropDelivered','billFailed'];
var SET_KEYS = ['nimex_config_v1','nimex_vkind_v1','nimex_vtype_v1','nimex_waive_v1','nimex_fine_toggles_v1','share'];
var CK = 'cid-data', CHUNK = 90000, PART = 45000, TMP = 'Rows_upload';

function ss_() { return SpreadsheetApp.getActiveSpreadsheet(); }
function tab_(name, head) {
  var t = ss_().getSheetByName(name);
  if (!t) { t = ss_().insertSheet(name); t.getRange(1, 1, 1, head.length).setValues([head]).setFontWeight('bold'); t.setFrozenRows(1); }
  return t;
}
function log_(a, d) { tab_('Log', ['time', 'action', 'detail']).appendRow([new Date(), a, d]); }
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

/* ---------- Settings: key | part | value (ข้อความ JSON) ---------- */
function readSettings_() {
  var t = tab_('Settings', ['key', 'part', 'value']), n = t.getLastRow(), parts = {}, o = {};
  if (n > 1) t.getRange(2, 1, n - 1, 3).getValues().forEach(function (r) {
    var k = String(r[0]); if (!k || SET_KEYS.indexOf(k) < 0) return;
    (parts[k] = parts[k] || [])[+r[1] || 0] = String(r[2]);
  });
  for (var k in parts) o[k] = parts[k].join('');
  return o;
}
function writeSettings_(o) {
  var t = tab_('Settings', ['key', 'part', 'value']), n = t.getLastRow(), rows = [];
  if (n > 1) t.getRange(2, 1, n - 1, 3).clearContent();
  Object.keys(o).forEach(function (k) { var s = String(o[k]); for (var i = 0, p = 0; i < s.length || p === 0; i += PART, p++) rows.push([k, p, s.substr(i, PART)]); });
  if (rows.length) { var r = t.getRange(2, 1, rows.length, 3); r.setNumberFormat('@'); r.setValues(rows); }
}

/* ---------- ข้อมูลสาธารณะ (แคช gzip+base64 แบ่งชิ้น 90 KB) ---------- */
function rowsOf_(t) { var n = t.getLastRow(); return n < 2 ? [] : t.getRange(2, 1, n - 1, COLS.length).getDisplayValues(); }
function publicJson_() {
  return JSON.stringify({ ok: true, v: 1, cols: COLS, rows: rowsOf_(tab_('Rows', COLS)), settings: readSettings_(),
    updated: PropertiesService.getScriptProperties().getProperty('UPDATED') || '' });
}
function cacheGet_() {
  var c = CacheService.getScriptCache(), n = c.get(CK + '-n'); if (!n) return null;
  var keys = []; for (var i = 0; i < +n; i++) keys.push(CK + '-' + i);
  var all = c.getAll(keys), b = '';
  for (i = 0; i < keys.length; i++) { if (!all[keys[i]]) return null; b += all[keys[i]]; }
  return Utilities.ungzip(Utilities.newBlob(Utilities.base64Decode(b), 'application/x-gzip')).getDataAsString('UTF-8');
}
function cachePut_(s) {
  var b = Utilities.base64Encode(Utilities.gzip(Utilities.newBlob(s, 'text/plain')).getBytes()), o = {}, n = 0;
  for (var i = 0; i < b.length; i += CHUNK) o[CK + '-' + (n++)] = b.substr(i, CHUNK);
  if (n <= 90) { o[CK + '-n'] = String(n); CacheService.getScriptCache().putAll(o, 21600); }
}
function cacheClear_() { CacheService.getScriptCache().remove(CK + '-n'); }
function refresh_() { cacheClear_(); try { cachePut_(publicJson_()); } catch (x) {} }

function doGet() {
  var s = null;
  try { s = cacheGet_(); } catch (e) {}
  if (!s) { s = publicJson_(); try { cachePut_(s); } catch (e) {} }
  return ContentService.createTextOutput(s).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var b;
  try { b = JSON.parse(e.postData.contents); } catch (x) { return json_({ ok: false, error: 'bad_request' }); }
  var key = PropertiesService.getScriptProperties().getProperty('ADMIN_KEY');
  if (!key) return json_({ ok: false, error: 'admin_key_not_set' });
  if (!b.key || b.key !== key) { Utilities.sleep(1200); return json_({ ok: false, error: 'denied' }); }
  if (b.a === 'auth') return json_({ ok: true });
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) return json_({ ok: false, error: 'busy' });
  try {
    if (b.a === 'settings') {        // { key: 'ข้อความ JSON' }  ค่า null = ลบ
      var cur = readSettings_(), inc = b.settings || {};
      for (var k in inc) {
        if (SET_KEYS.indexOf(k) < 0) return json_({ ok: false, error: 'bad_key', key: k });
        if (inc[k] === null) { delete cur[k]; continue; }
        if (typeof inc[k] !== 'string' || inc[k].length > 5000000) return json_({ ok: false, error: 'bad_value', key: k });
        try { JSON.parse(inc[k]); } catch (x) { return json_({ ok: false, error: 'bad_value', key: k }); }
        cur[k] = inc[k];
      }
      writeSettings_(cur); log_('settings', Object.keys(inc).join(', '));
      refresh_();
      return json_({ ok: true, settings: cur });
    }
    if (b.a === 'begin') {            // แท็บชั่วคราวสำหรับรับข้อมูลชุดใหม่ทั้งหมด
      var old = ss_().getSheetByName(TMP); if (old) ss_().deleteSheet(old);
      tab_(TMP, COLS);
      return json_({ ok: true });
    }
    if (b.a === 'append') {           // ไม่เกิน 4,000 แถวต่อครั้ง
      var t = ss_().getSheetByName(TMP); if (!t) return json_({ ok: false, error: 'not_begun' });
      var r = b.rows || [];
      if (!Array.isArray(r) || r.length > 5000) return json_({ ok: false, error: 'bad_rows' });
      if (r.length) {
        var rg = t.getRange(t.getLastRow() + 1, 1, r.length, COLS.length);
        rg.setNumberFormat('@');      // เก็บเป็นข้อความ: วันที่/เวลา/รหัสที่ขึ้นต้นด้วย 0 ไม่ถูกแปลง
        rg.setValues(r.map(function (x) { return COLS.map(function (c, i) { var v = x[i]; return v === null || v === undefined ? '' : String(v); }); }));
      }
      return json_({ ok: true, total: t.getLastRow() - 1 });
    }
    if (b.a === 'finish') {
      var nt = ss_().getSheetByName(TMP); if (!nt) return json_({ ok: false, error: 'not_begun' });
      var total = nt.getLastRow() - 1;
      if (typeof b.expect === 'number' && b.expect !== total) return json_({ ok: false, error: 'count_mismatch', got: total, expect: b.expect });
      var ot = ss_().getSheetByName('Rows'); if (ot) ss_().deleteSheet(ot);
      nt.setName('Rows');
      PropertiesService.getScriptProperties().setProperty('UPDATED', new Date().toISOString());
      log_(b.note || 'upload', total + ' rows');
      refresh_();
      return json_({ ok: true, total: total });
    }
    return json_({ ok: false, error: 'unknown' });
  } finally { lock.releaseLock(); }
}
